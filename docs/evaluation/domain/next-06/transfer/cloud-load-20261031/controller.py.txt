def event(name,**data):
    report['events'].append({'utc':datetime.now(UTC).strftime('%Y-%m-%d %H:%M:%S UTC'),'event':name,**data})
    (folder/'result.json').write_text(json.dumps(report,indent=2)+'\n')
    print(name,flush=True)

def call(service,operation,**kwargs):
    if report['api_calls']>=160: raise Refused('api_budget')
    if not cleaning and time.monotonic()>=deadline: raise Refused('work_deadline')
    report['api_calls']+=1
    return getattr(clients[service],operation)(**kwargs)

def idle():
    c=call('ecs','describe_clusters',clusters=[CLUSTER])
    s=call('ecs','describe_services',cluster=CLUSTER,services=[SERVICE])
    if c.get('failures') or s.get('failures') or len(c.get('clusters',[]))!=1 or len(s.get('services',[]))!=1:
        raise Refused('shared_snapshot_missing')
    cluster=c['clusters'][0]; service=s['services'][0]
    if cluster.get('runningTasksCount') or cluster.get('pendingTasksCount') or any(service.get(k) for k in ('desiredCount','runningCount','pendingCount')):
        raise Refused('shared_workload_active')

def db_state():
    values=call('rds','describe_db_instances',DBInstanceIdentifier=DB)['DBInstances']
    if len(values)!=1: raise Refused('database_ambiguous')
    d=values[0]
    if (d['DBInstanceIdentifier']!=DB or d['Engine']!='postgres' or d['EngineVersion']!='18.6'
        or d['DBInstanceClass']!='db.t4g.micro' or d['AllocatedStorage']!=20 or d['StorageType']!='gp3'
        or d['MultiAZ'] or d['PubliclyAccessible'] or not d['StorageEncrypted']
        or d['Endpoint']['Address']!=HOST or d['Endpoint']['Port']!=5432
        or {g['VpcSecurityGroupId'] for g in d['VpcSecurityGroups']}!={'sg-02f98707f8e43d2ea'}):
        raise Refused('database_binding_drift')
    return d['DBInstanceStatus']

def asg_guard():
    groups=call('autoscaling','describe_auto_scaling_groups',AutoScalingGroupNames=[ASG])['AutoScalingGroups']

    event('asg_safety_snapshot',desired=g['DesiredCapacity'],minimum=g['MinSize'],maximum=g['MaxSize'],
          instances=[{'id':i['InstanceId'],'lifecycle':i['LifecycleState']} for i in g['Instances']],suspended=sorted(suspended))
    if (g['DesiredCapacity']!=0 or g['MinSize']!=0 or g['MaxSize']!=1
        or {i['InstanceId'] for i in g['Instances']}!={INSTANCE}): raise Refused('asg_membership_or_capacity_drift')
    if not {'Launch','Terminate'} <= suspended: raise Refused('asg_automation_not_suspended')

def wait_for(label,read,predicate,seconds):
    until=min(time.monotonic()+seconds,deadline if not cleaning else cleanup_deadline)
    while True:
        value=read()
        if predicate(value): return value
        if time.monotonic()>=until: raise Refused(label+'_timeout')
        event('waiting_'+label,state=value)
        time.sleep(min(30,max(0,until-time.monotonic())))

def ssm_online():

    return len(rows)==1 and rows[0].get('PingStatus')=='Online'
