def event(name,**data):
    report['events'].append({'utc':datetime.now(UTC).strftime('%Y-%m-%d %H:%M:%S UTC'),'event':name,**data})
    (folder/'result.json').write_text(json.dumps(report,indent=2)+'\n')
    print(name,flush=True)

def call(service,operation,**kwargs):

    report['api_calls']+=1

def wait_for(label,read,predicate,seconds):
    until=min(time.monotonic()+seconds,deadline if not cleaning else cleanup_deadline)
    while True:
        value=read()
        if predicate(value): return value
        if time.monotonic()>=until: raise Refused(label+'_timeout')
        event('waiting_'+label,state=value)
        time.sleep(min(30,max(0,until-time.monotonic())))

    if value.get('snapshot_passed') is True:return 'snapshot_verified_resources_stopped'
    return 'snapshot_failed_resources_stopped'
